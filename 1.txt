        protected override void OnKeyPress(KeyPressEventArgs e)
        {
            base.OnKeyPress(e);

            if (e.KeyChar == ((char)Keys.F8))
            {
                var form = new Forms.UI.frmComboBoxItems(this);
                form.Closed += frmComboBoxItems_Closed;
                form.ShowDialog();
            }
        }

        void frmComboBoxItems_Closed(object sender, EventArgs e)
        {
            var form = sender as Forms.UI.frmComboBoxItems;

            if (form?.DialogResult == DialogResult.OK)
            {
                this.SetValue(form.SelectedValue);
            }
        }
